# Pinned to exact versions rather than floors (`>=`).
#
# These were previously all `>=`, which meant two installs a month apart could
# resolve to different libraries and behave differently — a real problem for a
# project whose evaluation numbers (tests/test_retrieval.py) depend on the
# embedding stack producing the same vectors it did when the thresholds were
# measured. A floating `sentence-transformers` or `chromadb` could shift those
# scores without a single line of this project changing.
#
# The versions below are the ones the working venv resolved to and that the
# measured baselines were taken against. To upgrade deliberately: bump a pin,
# reinstall, and re-run `pytest tests/test_retrieval.py` — if the retrieval
# metrics moved, that is the upgrade's effect, and it is now visible.
#
# Transitive dependencies are not pinned here. A full lockfile (pip-compile)
# would be stricter still; this is the level of reproducibility that is worth
# maintaining by hand for a project this size.

fastapi==0.141.1
uvicorn[standard]==0.52.3
pydantic==2.13.4
python-dotenv==1.2.3
